       344 +  
       345 +  ### 6.1 文件结构
       346 +  
       347 +  ```xml
       348 +  <?xml version="1.0" encoding="UTF-8"?>
       349 +  <svg xmlns="http://www.w3.org/2000/svg"
       350 +       xmlns:bc="http://bimcanvas.io/schema/1.0"
       351 +       width="8000" height="6000"
       352 +       bc:version="1.0"
       353 +       bc:canvas-id="canvas_001">
       354 +  
       355 +    <!-- 建筑结构层（锁定） -->
       356 +    <g id="layer-structure" bc:type="layer" bc:locked="true">
       357 +      <g bc:type="wall" bc:element-id="wall_001">...</g>
       358 +      <g bc:type="door" bc:element-id="door_001">...</g>
       359 +      <g bc:type="window" bc:element-id="window_001">...</g>
       360 +    </g>
       361 +  
       362 +    <!-- 家具层 -->
       363 +    <g id="layer-furniture" bc:type="layer">
       364 +      <g bc:type="furniture"
       365 +         bc:element-id="furniture_001"
       366 +         bc:family-id="sofa_001"
       367 +         bc:revit-type-id="337201"
       368 +         transform="translate(3000,2000) rotate(90)">
       369 +        <!-- SVG图形 -->
       370 +      </g>
       371 +    </g>
       372 +  
       373 +    <!-- 批注层 -->
       374 +    <g id="layer-annotation" bc:type="layer">
       375 +      <g bc:type="annotation" bc:element-id="anno_001">
       376 +        <text x="5000" y="3000">建议换成L型沙发</text>
       377 +      </g>
       378 +    </g>
       379 +  
       380 +  </svg>
       381 +  ```
